polovina bitrate davka pro Windows

@echo off
setlocal enabledelayedexpansion

set "INPUT=vstup.mp4"
set "OUTPUT=vystup_polovina.mp4"

:: Zjištění datového toku videa v bitech za sekundu pomocí MediaInfo
for /f "tokens=*" %%i in ('mediainfo --Inform^="Video;%%BitRate%%" "%INPUT%"') do set BITRATE=%%i

:: Pokud video nemá vyplněný přesný bitrate, zkusíme zjistit celkový bitrate souboru
if "%BITRATE%"=="" (
    for /f "tokens=*" %%i in ('mediainfo --Inform^="General;%%BitRate%%" "%INPUT%"') do set BITRATE=%%i
)

:: Pokud se nepodařilo zjistit nic, skript se ukončí s chybou
if "%BITRATE%"=="" (
    echo Chyba: Nepodarilo se zjistit bitrate souboru.
    pause
    exit /b
)

:: Výpočet poloviny původního bitratu
set /a HALF_BITRATE=%BITRATE% / 2

:: Spuštění ffmpeg s vypočítanou hodnotou
ffmpeg -i "%INPUT%" -b:v %HALF_BITRATE% -c:a copy "%OUTPUT%"

echo Puvodni bitrate: %BITRATE% bps
echo Novy bitrate: %HALF_BITRATE% bps
pause
